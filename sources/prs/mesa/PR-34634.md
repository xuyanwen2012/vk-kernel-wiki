SHADER compute compute_shader GLSL
#version 430
#extension GL_KHR_memory_scope_semantics : require

layout(set = 0, binding = 0) buffer coherent block1 {
   int temp[];

layout(set = 0, binding = 1) buffer block2 {
   int result[];
};

void main() {
   if (gl_LocalInvocationIndex < 64) {
      for (uint i = 0; i < 64; i++) {
         atomicAdd(result[gl_GlobalInvocationID.x], 1);
      }
   }

   temp[gl_GlobalInvocationID.x] = int(gl_GlobalInvocationID.x ^ 64);
   controlBarrier(gl_ScopeWorkgroup, gl_ScopeWorkgroup, gl_StorageSemanticsBuffer, gl_SemanticsAcquireRelease);
   result[gl_GlobalInvocationID.x] = temp[gl_GlobalInvocationID.x ^ 64] - int(gl_GlobalInvocationID.x);
}
END
